parser.py

def extract_code_sections(file_path): with open(file_path, 'r') as file: source = file.read()

tree = ast.parse(source)
code_sections = []

for node in ast.walk(tree):
    if isinstance(node, (ast.FunctionDef, ast.ClassDef)):
        start_line = node.lineno - 1
        end_line = max([getattr(n, 'lineno', start_line) for n in ast.walk(node)])
        code = "\n".join(source.splitlines()[start_line:end_line])
        code_sections.append(code)

return code_sections

quiz.py

def generate_quiz(code): lines = [line.strip() for line in code.splitlines() if line.strip() and not line.startswith("#")] questions = []

for line in lines:
    question = {
        "question": f"What does the following line do?\n{line}",
        "options": [line, "Raises an error", "Does nothing", "Creates a syntax error"],
        "answer": line
    }
    random.shuffle(question["options"])
    questions.append(question)

if not questions:
    questions.append({
        "question": "What is the purpose of this section of code?",
        "options": ["To perform input/output", "To define a function", "To import libraries", "None of the above"],
        "answer": "To define a function"
    })

return questions

        print("\nQuiz:", q["question"])
        for i, opt in enumerate(q["options"]):
            print(f"  {i + 1}. {opt}")
        try:
            user_input = int(input("Your answer (number): ")) - 1
            if q["options"][user_input] == q["answer"]:
                print("Correct!\n")
            else:
                print(f"Incorrect. Correct answer: {q['answer']}\n")
        except (ValueError, IndexError):
            print(f"Invalid input. Correct answer: {q['answer']}\n")
